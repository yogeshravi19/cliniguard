' PowerPoint VBA Macro Script
' Complete 13-Slide Auto-Generation Script
' ============================================================================
' INSTRUCTIONS: 
' 1. Open PowerPoint (new presentation)
' 2. Press Alt+F11 to open the VBA editor
' 3. Copy and paste this entire script into Module1
' 4. Run Sub CreateCLINIGUARDPresentation()
' 5. The macro will auto-generate all 13 professional slides

Option Explicit

' Define Color Scheme
Const COLOR_WHITE = RGB(248, 249, 250)        ' #F8F9FA
Const COLOR_DARK_BLUE = RGB(26, 54, 93)       ' #1A365D
Const COLOR_TEAL = RGB(13, 148, 136)          ' #0D9488
Const COLOR_LIGHT_GREY = RGB(243, 244, 246)   ' #F3F4F6
Const COLOR_TEXT_DARK = RGB(31, 41, 55)       ' #1F2937
Const COLOR_TEXT_LIGHT = RGB(107, 114, 128)   ' #6B7280
Const COLOR_ACCENT_RED = RGB(239, 68, 68)     ' #EF4444
Const COLOR_ACCENT_AMBER = RGB(251, 146, 60)  ' #FB923C
Const COLOR_ACCENT_GREEN = RGB(34, 197, 94)   ' #22C55E

Sub CreateCLINIGUARDPresentation()
    Dim prs As Presentation
    Set prs = ActivePresentation
    
    ' Clear existing slides if any
    While prs.Slides.Count > 0
        prs.Slides(1).Delete
    Wend
    
    ' Create all 13 slides
    Call CreateSlide1_Title(prs)
    Call CreateSlide2_ClinicalContext(prs)
    Call CreateSlide3_TheDanger(prs)
    Call CreateSlide4_TechnicalGaps(prs)
    Call CreateSlide5_ProposedFramework(prs)
    Call CreateSlide6_Module1_MultiSignal(prs)
    Call CreateSlide7_Module2_Fusion(prs)
    Call CreateSlide8_Module3_Explainability(prs)
    Call CreateSlide9_Module4_Mitigation(prs)
    Call CreateSlide10_Evaluation(prs)
    Call CreateSlide11_QuantitativeResults(prs)
    Call CreateSlide12_ClinicianValidation(prs)
    Call CreateSlide13_Conclusion(prs)
    
    MsgBox "CLINIGUARD presentation created successfully with 13 slides!", vbInformation
End Sub
